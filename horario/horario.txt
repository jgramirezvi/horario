server {
    listen 80;
    # Reemplaza 'tu_dominio.com' con tu nombre de dominio o la IP pública de tu servidor.
    server_name tu_dominio.com www.tu_dominio.com;

    location = /favicon.ico { access_log off; log_not_found off; }

    # Servir archivos estáticos directamente.
    # Reemplaza '/ruta/a/tu/proyecto/horario/staticfiles' con la ruta absoluta
    # a tu directorio STATIC_ROOT después de ejecutar `python manage.py collectstatic`.
    location /static/ {
        alias /ruta/a/tu/proyecto/horario/staticfiles/;
    }

    # Opcional: Servir archivos multimedia (subidos por usuarios).
    # Reemplaza '/ruta/a/tu/proyecto/horario/media' con la ruta absoluta a tu MEDIA_ROOT.
    location /media/ {
        alias /ruta/a/tu/proyecto/horario/media/;
    }

    # Pasar todas las demás peticiones a Gunicorn.
    location / {
        include proxy_params;
        # Estas cabeceras le dicen a tu app Django información sobre la petición original.
        proxy_set_header Host $http_host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;

        # La dirección donde Gunicorn está escuchando.
        # Coincide con `gunicorn --bind 127.0.0.1:8000 ...`
        proxy_pass http://127.0.0.1:8000;
    }
}
